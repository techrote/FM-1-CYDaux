cmake_minimum_required(VERSION 3.20)
project(FM1CYDaux LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}")
endforeach()
add_library(fm1core src/controller.cpp src/midi.cpp src/profiles.cpp src/sloop_protocol.cpp src/ui.cpp src/touch.cpp src/companion.cpp src/performance.cpp)
target_include_directories(fm1core PUBLIC include)
if(MSVC)
    target_compile_options(fm1core PRIVATE /W4)
else()
    target_compile_options(fm1core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_library(fm1virtual simulator/virtual_instrument.cpp
    third_party/zaaggenz/src/dsp.c third_party/zaaggenz/src/engine.c
    third_party/zaaggenz/src/presets.c third_party/zaaggenz/src/source_bank.c
    third_party/xenoresonator/xeno.c third_party/grimoire/orbit.c)
target_include_directories(fm1virtual PUBLIC simulator third_party/zaaggenz/include third_party/xenoresonator third_party/grimoire)
target_link_libraries(fm1virtual PUBLIC fm1core)
add_executable(fm1-simulator simulator/main.cpp)
target_link_libraries(fm1-simulator PRIVATE fm1virtual)
add_executable(fm1-resource-probe simulator/resource_probe.cpp)
target_link_libraries(fm1-resource-probe PRIVATE fm1virtual)
enable_testing()
add_executable(virtual_test simulator/virtual_test.cpp)
target_link_libraries(virtual_test PRIVATE fm1virtual)
if(MSVC)
    target_compile_options(virtual_test PRIVATE /UNDEBUG)
else()
    target_compile_options(virtual_test PRIVATE -UNDEBUG)
endif()
add_test(NAME virtual_test COMMAND virtual_test)
file(GLOB TEST_SOURCES CONFIGURE_DEPENDS "tests/*_test.cpp")
foreach(source IN LISTS TEST_SOURCES)
    get_filename_component(test "${source}" NAME_WE)
    add_executable(${test} "${source}")
    if(test STREQUAL "companion_controller_test" OR test STREQUAL "companion_gesture_test")
        target_link_libraries(${test} PRIVATE fm1virtual)
    else()
        target_link_libraries(${test} PRIVATE fm1core)
    endif()
    if(MSVC)
        target_compile_options(${test} PRIVATE /UNDEBUG)
    else()
        target_compile_options(${test} PRIVATE -UNDEBUG)
    endif()
    add_test(NAME ${test} COMMAND ${test})
endforeach()
