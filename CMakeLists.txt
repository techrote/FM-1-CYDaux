cmake_minimum_required(VERSION 3.20)
project(FM1CYDaux LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}")
endforeach()
add_library(fm1core src/controller.cpp src/midi.cpp src/profiles.cpp src/sloop_protocol.cpp src/ui.cpp src/touch.cpp)
target_include_directories(fm1core PUBLIC include)
if(MSVC)
    target_compile_options(fm1core PRIVATE /W4)
else()
    target_compile_options(fm1core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(fm1-simulator simulator/main.cpp)
target_link_libraries(fm1-simulator PRIVATE fm1core)
enable_testing()
file(GLOB TEST_SOURCES CONFIGURE_DEPENDS "tests/*_test.cpp")
foreach(source IN LISTS TEST_SOURCES)
    get_filename_component(test "${source}" NAME_WE)
    add_executable(${test} "${source}")
    target_link_libraries(${test} PRIVATE fm1core)
    if(MSVC)
        target_compile_options(${test} PRIVATE /UNDEBUG)
    else()
        target_compile_options(${test} PRIVATE -UNDEBUG)
    endif()
    add_test(NAME ${test} COMMAND ${test})
endforeach()
